            }
        } else {
            quizFinished = true;
            renderResult();
        }
    }

    // 直接跳到结果（提前结束）
    function jumpToResult() {
        const unanswered = userAnswers.filter(a => a === null).length;
        if (unanswered > 0 && !confirm(`还有 ${unanswered} 题未答，提前结束将按0分计算，确定吗？`)) return;
        quizFinished = true;
        renderResult();
    }

    function renderQuiz() {
        const total = QUESTIONS.length;
        const currentQ = QUESTIONS[currentIndex];
        const currentAnswer = userAnswers[currentIndex];
        const answeredCount = userAnswers.filter(a => a !== null).length;
        const progressPercent = ((currentIndex + 1) / total) * 100;

        let optionsHtml = '';
        OPTIONS.forEach(opt => {
            const isSelected = (currentAnswer === opt.label);
            const selectedClass = isSelected ? 'selected' : '';
            optionsHtml += `
                <div class="option ${selectedClass}" data-choice="${opt.label}">
                    <div class="option-letter">${opt.label}</div>
                    <div class="option-text">${opt.text}</div>
                </div>
            `;
        });

        const isLast = currentIndex === total - 1;
        const submitButton = isLast ? `<button class="primary" id="submitExamBtn" style="flex:1;">✅ 提交试卷</button>` : '';

        const html = `
            <div class="hero">
                <h1>🧸 父母资格证考试</h1>
                <div class="sub">测测你是神级父母 or 需要回炉重造？</div>
                <div class="badge">
                    <span>📋 ${answeredCount}/${total} 题已答</span>
                    <span>⚡ 真实回答 解锁段位</span>
                </div>
                <div class="progress-area">
                    <div class="progress-bar" style="width: ${progressPercent}%;"></div>
                </div>
                <div class="counter">第 ${currentIndex+1} / ${total} 题</div>
            </div>
            <div class="question-card">
                <div class="q-num">灵魂拷问 #${currentIndex+1}</div>
                <div class="q-text">${escapeHtml(currentQ)}</div>
                <div class="options" id="optionsContainer">
                    ${optionsHtml}
                </div>
                <div class="nav-buttons">
                    <button id="prevBtn" ${currentIndex === 0 ? 'disabled' : ''}>◀ 上一题</button>
                    <button id="nextBtn" class="primary">下一题 ▶</button>
                    ${submitButton}
                </div>
                <div style="margin-top: 18px; text-align:center;">
                    <button id="jumpFinishBtn" style="background:#d9c2ac; padding:10px 20px; font-size:0.85rem;">⚡ 直接查看结果</button>
                </div>
            </div>
            <footer>选项越靠近“总是这样”得分越高，段位越低 → 要警惕咯</footer>
        `;
        appContainer.innerHTML = html;

        // 绑定选项点击
        document.querySelectorAll('.option').forEach(el => {
            el.addEventListener('click', (e) => {
                const choice = el.getAttribute('data-choice');
                setCurrentAnswer(choice);
            });
        });

        const prevBtn = document.getElementById('prevBtn');
        if (prevBtn) prevBtn.addEventListener('click', prevQuestion);
        const nextBtn = document.getElementById('nextBtn');
        if (nextBtn) nextBtn.addEventListener('click', nextQuestion);
        const submitBtn = document.getElementById('submitExamBtn');
        if (submitBtn) submitBtn.addEventListener('click', finishExam);
        const jumpBtn = document.getElementById('jumpFinishBtn');
        if (jumpBtn) jumpBtn.addEventListener('click', jumpToResult);
    }

    function escapeHtml(str) {
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        }).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, function(c) {
            return c;
        });
    }

    function init() {
        renderQuiz();
    }
    init();
</script>
